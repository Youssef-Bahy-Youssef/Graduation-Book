% ---- Body text: 12pt (documentclass), justified, 1.5 line spacing ----
\justifying
\onehalfspacing
\setlength{\emergencystretch}{3em}
\uchyph=1
\hbadness=2000

% ---- Chapter titles: bold, 24pt, centered, "Chapter N:" prefix, horizontal rule ----
\titleformat{\chapter}[display]
  {\normalfont\fontsize{24}{28}\bfseries\centering}
  {\chaptertitlename\ \thechapter:}
  {0.75em}
  {{\hrule width \textwidth height 0.5pt}\vspace{0.5em}}
\titlespacing*{\chapter}{0pt}{0pt}{24pt}

% ---- Chapter* titles (unnumbered): centered, bold, horizontal rule ----
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\fontsize{24}{28}\bfseries\centering}
  {}
  {0pt}
  {{\hrule width \textwidth height 0.5pt}\vspace{0.5em}}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{24pt}

% ---- Section titles: bold, 14pt, left-aligned ----
\titleformat{\section}
  {\normalfont\fontsize{14}{17}\bfseries}
  {\thesection.}
  {0.75em}
  {}
\titlespacing*{\section}{0pt}{14pt}{8pt}

% ---- Subsection titles: bold, 12pt ----
\titleformat{\subsection}
  {\normalfont\fontsize{12}{14}\bfseries}
  {\thesubsection.}
  {0.75em}
  {}
\titlespacing*{\subsection}{0pt}{10pt}{6pt}

% ---- Figure captions below; table captions above (Figure X.Y / Table X.Y) ----
\captionsetup[figure]{
  position=bottom,
  labelfont=bf,
  labelsep=colon,
  font=small
}
\captionsetup[table]{
  position=top,
  labelfont=bf,
  labelsep=colon,
  font=small
}

% ---- References: 10pt ----
\renewcommand*{\bibfont}{\fontsize{10}{12}\selectfont}

% ---- Headers & footers: chapter name in header, page number centered in footer ----
\setlength{\headheight}{28pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[C]{\leftmark}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

\renewcommand{\chaptermark}[1]{%
  \ifnum\value{chapter}>0\relax
    \markboth{\MakeUppercase{\chaptername\ \thechapter: #1}}{}%
  \else
    \markboth{\MakeUppercase{#1}}{}%
  \fi
}

% Plain style for chapter opening pages (footer page number only)
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}
