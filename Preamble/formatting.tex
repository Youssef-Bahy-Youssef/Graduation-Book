% ---- Body text: 12pt (documentclass), justified, 1.5 line spacing ----
\justifying
\onehalfspacing
\setlength{\emergencystretch}{1.5em}
\uchyph=1
\hbadness=2000

% ---- Chapter titles: bold, 24pt, centered, "Chapter N:" prefix ----
\titleformat{\chapter}[display]
  {\normalfont\fontsize{24}{28}\bfseries\centering}
  {Chapter \thechapter:}
  {0.75em}
  {}
\titlespacing*{\chapter}{0pt}{0pt}{24pt}

% ---- Section titles: bold, 18pt, left-aligned ----
\titleformat{\section}
  {\normalfont\fontsize{18}{22}\bfseries}
  {\thesection.}
  {0.75em}
  {}
\titlespacing*{\section}{0pt}{18pt}{12pt}

% ---- Subsection titles: bold, 14pt ----
\titleformat{\subsection}
  {\normalfont\fontsize{14}{17}\bfseries}
  {\thesubsection.}
  {0.75em}
  {}
\titlespacing*{\subsection}{0pt}{12pt}{8pt}

% ---- Figure captions below; table captions above (Figure X.Y / Table X.Y) ----
\captionsetup[figure]{
  position=bottom,
  labelfont=bf,
  labelsep=colon,
  font=small
}
\captionsetup[table]{
  position=top,
  labelfont=bf,
  labelsep=colon,
  font=small
}
\captionsetup[sub]{
  font=small
}

% ---- References: 10pt ----
\renewcommand*{\bibfont}{\fontsize{10}{12}\selectfont}

% ---- Headers & footers: chapter name in header, page number centered in footer ----
\pagestyle{fancy}
\fancyhf{}
\fancyhead[C]{\leftmark}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}

\renewcommand{\chaptermark}[1]{%
  \markboth{\MakeUppercase{\chaptername\ \thechapter: #1}}{}%
}

% Plain style for chapter opening pages (footer page number only)
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

  urlcolor=CUGreen,
  filecolor=CUGreen,
